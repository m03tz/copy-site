```typescript
if (patientProfile?.email) {

    appointmentId: appointment.id as string,
    patientEmail: patientProfile.email,
    patientName: patientProfile.full_name_ar,
    scheduledStart: scheduled_start,
    appointmentType: appointment_type,
  })

  // Store Resend email IDs for potential cancellation
  if (reminder24hId || reminder2hId) {
    await supabase
      .from('appointments')
      .update({
        reminder_24h_email_id: reminder24hId,
        reminder_2h_email_id: reminder2hId,
      })
      .eq('id', appointment.id as string)
  }
}
// If no email, skip silently — do not fail the booking
```

```typescript

  // Cancel any scheduled reminder emails

    appointment.reminder_24h_email_id,
    appointment.reminder_2h_email_id
  )

  // Send cancellation notification to patient
  const { data: patientProfile } = await supabase
    .from('profiles')

      patientName: patientProfile.full_name_ar,

    })
  }

  // Email operations failed — cancellation still succeeded

}
```
